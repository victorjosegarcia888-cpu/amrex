11. ECUACIONES PARA SCRIPTS FFSC
-----------------------------------------------------------

// Mezcla
O_F = mdot_ox / mdot_fuel = 3.6

// Presión de cámara
Pc = 330e5

// Empuje
F = mdot_total * Ve + (Pe - Pa) * Ae

// Isp
Isp = Ve / g0

// Perfil Bell
R(x) = a0 + a1*x + a2*x^2 + a3*x^3

// Canal helicoidal
r(t) = R * cos(t)
z(t) = pitch * t
