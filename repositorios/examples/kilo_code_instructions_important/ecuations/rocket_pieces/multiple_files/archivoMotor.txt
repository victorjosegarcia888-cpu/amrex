using System;
using System.Numerics;
using PicoGK;
using LEAP71.ShapeKernel;
using LEAP71.LatticeLibrary;
using LEAP71.QuasiCrystals;
using LEAP71.HelixHeatX;

namespace RocketEngineDesign
{
    public static class EngineTasks
    {
        // ============================================================
        // 1 — Turbina Vectorial XA103
        // ============================================================
        public static Voxels Task_GenerateVectorTurbine()
        {
            var vox = new Voxels(0.25);
            int bladeCount = 26;
            double R = 0.1154;
            double h = 0.012;

            for (int i = 0; i < bladeCount; i++)
            {
                double angle = (Math.PI * 2.0 / bladeCount) * i;
                var blade = ShapeKernel.CreateAxialBlade(R, h, 0.002, 0.15);
                blade.RotateZ(angle);
                vox.Add(blade);
            }

            return vox;
        }

        // ============================================================
        // 2 — Impulsor de Bomba ASE (Helical Inducer NASA TM X‑419)
        // ============================================================
        public static Voxels Task_GenerateFuelPumpImpeller()
        {
            var vox = new Voxels(0.25);
            double Din = 0.045;
            double Dout = 0.134;
            double h = 0.0025;
            int bladeCount = 12;

            for (int i = 0; i < bladeCount; i++)
            {
                double angle = (Math.PI * 2.0 / bladeCount) * i;
                var blade = ShapeKernel.CreateHelicalBlade(Din * 0.5, Dout * 0.5, h, 0.0015, 83.0);
                blade.RotateZ(angle);
                vox.Add(blade);
            }

            var hub = ShapeKernel.CreateCylinder(Din * 0.3, h * 1.2);
            vox.Subtract(hub);
            return vox;
        }

        // ============================================================
        // 3 — Prequemador Toroidal ASE
        // ============================================================
        public static Voxels Task_GeneratePreburnerToroid()
        {
            var vox = new Voxels(0.25);
            var toroid = ShapeKernel.CreateTorus(0.06, 0.01);
            var inner = ShapeKernel.CreateCylinder(0.05, 0.12);
            vox.Add(toroid);
            vox.Subtract(inner);
            return vox;
        }

        // ============================================================
        // 4 — Cámara de Combustión Híbrida (Noyron TKL‑200)
        // ============================================================
        public static Voxels Task_GenerateHybridChamber()
        {
            var vox = new Voxels(0.25);
            double Dc = 0.380;
            double Dt = 0.198;
            double De = 1.060;

            var profile = ShapeKernel.CreateBellNozzleProfile(Dc, Dt, De, 25.0, 34.0, 10.5);
            var nozzle = ShapeKernel.SweepProfile(profile, 1.2);
            vox.Add(nozzle);

            var core = ShapeKernel.CreateCylinder(Dt * 0.5, 1.2);
            vox.Subtract(core);
            return vox;
        }

        // ============================================================
        // 5 — Lattice Aperiodico Interno (QuasiCrystals)
        // ============================================================
        public static Voxels Task_GenerateAperiodicLattice()
        {
            var vox = new Voxels(0.25);
            var qc = QuasiCrystals.CreateQuasicrystalField(10, 0.015, 8.0);
            var core = ShapeKernel.CreateCylinder(0.10, 0.30);
            vox.Add(core);
            vox.Modulate(qc);
            return vox;
        }

        // ============================================================
        // 6 — Canales Regenerativos HelixHeatX
        // ============================================================
        public static Voxels Task_GenerateCoolingChannels()
        {
            var vox = new Voxels(0.25);
            int channelCount = 120;

            for (int i = 0; i < channelCount; i++)
            {
                double angle = (Math.PI * 2.0 / channelCount) * i;
                var channel = HelixHeatX.CreateHelicalChannel(0.001, 0.001, 35.0, 0.15);
                channel.RotateZ(angle);
                vox.Add(channel);
            }

            return vox;
        }

        // ============================================================
        // 7 — Tobera de Plasma Negra (UHTC)
        // ============================================================
        public static Voxels Task_GeneratePlasmaNozzle()
        {
            var vox = new Voxels(0.25);
            var nozzle = ShapeKernel.CreateBellNozzleProfile(0.198, 0.106, 0.62, 25.0, 34.0, 10.5);
            var shell = ShapeKernel.SweepProfile(nozzle, 0.8);
            vox.Add(shell);
            return vox;
        }

        // ============================================================
        // 8 — Electrodos de Ionización
        // ============================================================
        public static Voxels Task_GenerateIonizationElectrodes()
        {
            var vox = new Voxels(0.25);
            int electrodeCount = 8;
            double radius = 0.05;

            for (int i = 0; i < electrodeCount; i++)
            {
                double angle = (Math.PI * 2.0 / electrodeCount) * i;
                var electrode = ShapeKernel.CreateCylinder(0.002, 0.05);
                electrode.RotateZ(angle);
                electrode.Translate(radius * Math.Cos(angle), radius * Math.Sin(angle), 0.0);
                vox.Add(electrode);
            }

            return vox;
        }

        // ============================================================
        // 9 — Tobera de Escape
        // ============================================================
        public static Voxels Task_GenerateExhaustNozzle()
        {
            var vox = new Voxels(0.25);
            var exit = ShapeKernel.CreateCone(0.06, 0.12, 0.20);
            vox.Add(exit);
            return vox;
        }
    }
}
⚙️ Program.cs — Ensamblado final del motor híbrido compacto
csharp
using System;
using PicoGK;
using RocketEngineDesign;

class Program
{
    static void Main(string[] args)
    {
        Library.Go(0.25);
        var engine = new Voxels(0.25);

        // Ensamblaje de componentes principales
        engine.Add(EngineTasks.Task_GenerateVectorTurbine());
        engine.Add(EngineTasks.Task_GenerateFuelPumpImpeller());
        engine.Add(EngineTasks.Task_GeneratePreburnerToroid());
        engine.Add(EngineTasks.Task_GenerateHybridChamber());
        engine.Add(EngineTasks.Task_GenerateAperiodicLattice());
        engine.Subtract(EngineTasks.Task_GenerateCoolingChannels());
        engine.Add(EngineTasks.Task_GeneratePlasmaNozzle());
        engine.Add(EngineTasks.Task_GenerateIonizationElectrodes());
        engine.Add(EngineTasks.Task_GenerateExhaustNozzle());

        // Filtros morfológicos
        engine.DoOffset(+0.003);
        engine.DoOffset(-0.003);
        engine.DoGaussianBlur(1.0);

        // Exportación
        var mesh = new Mesh(engine);
        Library.ExportMeshToStl(mesh, "output/Noyron_TKL200_Compact.stl");
        Library.ExportMeshTo3MF(mesh, "output/Noyron_TKL200_Compact.3mf");

        Console.WriteLine("Motor híbrido compacto Noyron TKL‑200 generado y exportado correctamente.");
    }
}
